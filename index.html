Connect to Google Sheets

All your bookkeeping data saves to and loads from Google Sheets automatically — same as the management dashboard.

One-time setup:

  1. Open "Foundation Living — Bookkeeping Data 2025-2026" in Drive
  2. Click Extensions → Apps Script
  3. Delete everything and paste the code below
  4. Click Deploy → New deployment → Web app
  5. Set Execute as: Me  |  Who has access: Anyone
  6. Deploy, authorize, copy the Web App URL, paste it below
function doGet(e){ const ss=SpreadsheetApp.getActiveSpreadsheet(); const inc=ss.getSheetByName('INCOME'); const exp=ss.getSheetByName('EXPENSES'); return ContentService.createTextOutput(JSON.stringify({ income:sheetToJSON(inc),expenses:sheetToJSON(exp), lastSaved:PropertiesService.getScriptProperties().getProperty('lastSaved')||'' })).setMimeType(ContentService.MimeType.JSON); } function doPost(e){ const d=JSON.parse(e.postData.contents); const ss=SpreadsheetApp.getActiveSpreadsheet(); if(d.income) writeSheet(ss,'INCOME',['month','client','invoices','amount','status'],d.income); if(d.expenses) writeSheet(ss,'EXPENSES',['date','description','category','status','amount'],d.expenses); PropertiesService.getScriptProperties().setProperty('lastSaved',new Date().toISOString()); return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON); } function sheetToJSON(sheet){ if(!sheet) return []; const rows=sheet.getDataRange().getValues(); if(rows.length<2) return []; const h=rows[0].map(x=>String(x).trim().toLowerCase()); return rows.slice(1).filter(r=>r.some(c=>c!=='')).map(r=>{const o={};h.forEach((k,i)=>{o[k]=r[i];});return o;}); } function writeSheet(ss,name,headers,data){ let sh=ss.getSheetByName(name); if(!sh) sh=ss.insertSheet(name); sh.clearContents(); sh.appendRow(headers.map(h=>h.toUpperCase())); data.forEach(r=>sh.appendRow(headers.map(k=>r[k]||''))); }
